# Plans de tournage de la vidéo promo (rendus avec --write-movie à 30 i/s)
var m

func run_id() -> int:
	for a in OS.get_cmdline_user_args():
		if a.begins_with("run="): return int(a.substr(4))
	return 1

func mark(name: String) -> void: print("SHOT_START ", name, " ", Engine.get_frames_drawn())

# —— outils caméra ——
func path_cam(a: Vector3, b: Vector3, la: Vector3, lb: Vector3, dur: float) -> Callable:
	var st := {"t": 0.0}
	return func(dt: float):
		st.t += dt
		var k: float = smoothstep(0.0, 1.0, clamp(st.t / dur, 0.0, 1.0))
		m.cam.global_position = a.lerp(b, k); m.cam.look_at(la.lerp(lb, k), Vector3.UP)

func follow_cam(off0: Vector3, off1: Vector3, dur: float, look_up := 1.0) -> Callable:
	var st := {"t": 0.0, "p": Vector3.INF}
	return func(dt: float):
		st.t += dt
		var k: float = smoothstep(0.0, 1.0, clamp(st.t / dur, 0.0, 1.0))
		var tp: Vector3 = m.player.global_position
		st.p = tp if st.p == Vector3.INF else st.p.lerp(tp, 1.0 - exp(-dt * 6.0))
		m.cam.global_position = st.p + off0.lerp(off1, k); m.cam.look_at(st.p + Vector3(0, look_up, 0), Vector3.UP)

func orbit_cam(c: Vector3, r: float, h: float, a0: float, a1: float, dur: float, look_h := 1.0) -> Callable:
	var st := {"t": 0.0}
	return func(dt: float):
		st.t += dt
		var a: float = lerp(a0, a1, clamp(st.t / dur, 0.0, 1.0))
		m.cam.global_position = c + Vector3(cos(a) * r, h, sin(a) * r); m.cam.look_at(c + Vector3(0, look_h, 0), Vector3.UP)

func nocull(n: Node) -> void:
	if n is GeometryInstance3D: (n as GeometryInstance3D).visibility_range_end = 0.0
	for c in n.get_children(): nocull(c)

func cinema(on: bool) -> void:
	m.hud.visible = not on
	for t in m.hud.toasts.get_children(): t.queue_free()

func gear() -> void:
	Game.S.tips["start"] = 1
	Game.S.gear = {"hache": 5, "pioche": 5, "faucille": 5, "epee": 5, "armure": 5, "bottes": 5, "bouclier": 5, "artefact": 5, "monture": 5}
	Game.S.weapon_kind = "epee"; Game.S.armor_kind = "plate"; Game.S.mount_kind = "pegase"; Game.S.artefact_kind = "rage"
	Game.S.ench = {"epee": 4, "armure": 3, "bouclier": 3, "bottes": 2}
	Game.S.wxp["epee"] = {"lvl": 30, "xp": 0}
	Game.S.silver = 48500000; Game.S.potions = 12
	m.player.refresh_gear(); m.player.hp = m.player.max_hp

func setup_common() -> void:
	m.hud.close_panel(); gear()
	m.cam.far = 420.0; m.sun.directional_shadow_max_distance = 90.0
	nocull(m.world)
	m.arrow.scale = Vector3.ONE * 0.001; m.beacon.scale = Vector3.ONE * 0.001
	m.next_boss_at = 99999.0
	m.hud.fps_lbl.visible = false

func put(p: Vector3) -> void:
	m.player.global_position = Vector3(p.x, m.world.height(p.x, p.z) + 0.4, p.z); m.player.velocity = Vector3.ZERO

func plan(main) -> Array:
	m = main
	match run_id():
		2: return plan2()
		3: return plan3()
		4: return plan4()
		5: return plan5()
	return plan1()
